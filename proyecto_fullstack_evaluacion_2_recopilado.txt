import { useState, useEffect } from "react";
import Home from "./pages/Home";
import Contacto from "./pages/Contacto";
import { productosIniciales } from "./data/productos";
import Productos from "./pages/Productos";
import Login from "./pages/Login";
import Registro from "./pages/Registro";
import Blog from "./pages/Blog";
import AdminHome from "./pages/AdminHome";
import AdminUsuarios from "./pages/AdminUsuarios";
import LayoutPrincipal from "./layouts/LayoutPrincipal";
import { Routes, Route } from "react-router-dom";
import RutaProtegida from "./components/RutaProtegida";
import { calcularCantidadTotal } from "./utils/carrito.js"


function App() {
  // cargar el carrito desde localstorage
  const [carrito, setCarrito] = useState(() => {
    const guardado = localStorage.getItem("carrito");
    return guardado ? JSON.parse(guardado) : [];
  });

  // guardado automático en localstorage cuando haya cambios
  useEffect(() => {
    localStorage.setItem("carrito", JSON.stringify(carrito));
  }, [carrito]);


  const [productos, setProductos] = useState(() => {
    const guardados = localStorage.getItem("productos");
    return guardados ? JSON.parse(guardados) : productosIniciales;
  });

  useEffect(() => {
    localStorage.setItem("productos", JSON.stringify(productos));
  }, [productos]);

  // agregar al carrito (max 5)
  function agregarAlCarrito(id) {
    setCarrito((carritoActual) => {
      const itemExistente = carritoActual.find((item) => item.id === id);

      if (itemExistente) {
        if (itemExistente.cantidad >= 5) {
          alert("Máximo 5 unidades por producto");
          return carritoActual;
        }
        return carritoActual.map((item) =>
          item.id === id ? { ...item, cantidad: item.cantidad + 1 } : item
        );
      } else {
        const producto = productosIniciales.find((p) => p.id === id);
        return [...carritoActual, { ...producto, cantidad: 1 }];
      }
    });
  }

  // disminuir
  function disminuirCantidad(id) {
    setCarrito((carritoActual) => {
      const item = carritoActual.find((i) => i.id === id);
      if (!item) return carritoActual;

      if (item.cantidad <= 1) {
        return carritoActual.filter((i) => i.id !== id);
      } else {
        return carritoActual.map((i) =>
          i.id === id ? { ...i, cantidad: item.cantidad - 1 } : i
        );
      }
    });
  }

  // eliminar
  function eliminarDelCarrito(id) {
    setCarrito((carritoActual) =>
      carritoActual.filter((item) => item.id !== id)
    );
  }

  // vaciar carrito
  function vaciarCarrito() {
    setCarrito([]);
  }

  // comprar (prueba no más)
  function comprar() {
    if (carrito.length === 0) {
      alert("El carrito está vacío.");
      return;
    }
    alert("¡Gracias por tu compra en HuertoHogar!");
    vaciarCarrito();
  }

  // total de productos
  const cantidadTotalCarrito = calcularCantidadTotal(carrito);  
  
  return (
    <Routes>
      {/* vistas públicas */}
      <Route element={<LayoutPrincipal cantidadCarrito={cantidadTotalCarrito} />}>
        <Route
          index
          element={
            <Home
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
        <Route
          path="productos"
          element={
            <Productos
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
        <Route
          path="contacto"
          element={
            <Contacto
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
        <Route
          path="login"
          element={
            <Login
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
        <Route
          path="registro"
          element={
            <Registro
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
        <Route
          path="blog"
          element={
            <Blog
              carrito={carrito}
              vaciarCarrito={vaciarCarrito}
              comprar={comprar}
              agregarAlCarrito={agregarAlCarrito}
              disminuirCantidad={disminuirCantidad}
              eliminarDelCarrito={eliminarDelCarrito}
            />
          }
        />
      </Route>{/* vistas admin */}
        <Route
          path="admin"
          element={
            <RutaProtegida exigirAdmin={true}>
              <AdminHome
                productos={productos}
                setProductos={setProductos}
              />
            </RutaProtegida>
          }
        />
        <Route
          path="admin/usuarios"
          element={
            <RutaProtegida exigirAdmin={true}>
              <AdminUsuarios />
            </RutaProtegida>
          }
        />
    </Routes>
  );
}

export default App;

const CLAVE_SESION = "usuarioActual";

/**
 * valida credenciales e inicia sesión.
 * soporta las credenciales oficiales de huertohogar (admin@duoc.cl / admin)
 * y las didácticas del tutorial docente (admin / admin123).
 */
export function iniciarSesion(usuario, password) {
  const usuarioLimpio = usuario.trim();
  const passwordLimpia = password.trim();

  // caso 1: administrador
  if (
    (usuarioLimpio === "admin@duoc.cl" && passwordLimpia === "admin") ||
    (usuarioLimpio === "admin" && passwordLimpia === "admin123")
  ) {
    const sesion = { usuario: usuarioLimpio, rol: "ADMIN" };
    localStorage.setItem(CLAVE_SESION, JSON.stringify(sesion));
    return { exito: true, rol: "ADMIN", sesion };
  }

  // caso 2: cliente general válido (según reglas de dominio institucional o gmail)
  const esCorreoValido =
    usuarioLimpio.endsWith("@duoc.cl") ||
    usuarioLimpio.endsWith("@profesor.duoc.cl") ||
    usuarioLimpio.endsWith("@gmail.com");

  if (esCorreoValido && passwordLimpia.length >= 4 && passwordLimpia.length <= 10) {
    const sesion = { usuario: usuarioLimpio, rol: "CLIENTE" };
    localStorage.setItem(CLAVE_SESION, JSON.stringify(sesion));
    return { exito: true, rol: "CLIENTE", sesion };
  }

  return { exito: false, error: "Credenciales incorrectas." };
}

/**
 * obtiene el objeto de sesión almacenado en localstorage.
 */
export function obtenerSesion() {
  try {
    const datos = localStorage.getItem(CLAVE_SESION);
    return datos ? JSON.parse(datos) : null;
  } catch (error) {
    console.error("Error al leer sesión:", error);
    return null;
  }
}

/**
 * elimina la sesión activa de localstorage.
 */
export function cerrarSesion() {
  localStorage.removeItem(CLAVE_SESION);
}

/**
 * revisa si existe un usuario autenticado
 */
export function estaAutenticado() {
  return obtenerSesion() !== null;
}

/**
 * revisa si el usuario autenticado tiene rol de admin
 */
export function esAdmin() {
  const sesion = obtenerSesion();
  return sesion?.rol === "ADMIN";
}

/* cálculo del monto total a pagar de los productos en el carrito */
export function calcularTotal(carrito) {
  if (!Array.isArray(carrito)) return 0;
  return carrito.reduce(
    (total, item) => total + (item.precio || 0) * (item.cantidad || 0),
    0
  );
}

/* cálculo de la cantidad total de unidades en el carrito. */
export function calcularCantidadTotal(carrito) {
  if (!Array.isArray(carrito)) return 0;
  return carrito.reduce(
    (total, item) => total + (item.cantidad || 0),
    0
  );
}

/* validación de un texto según longitud dsps de eliminar espacios laterales. */
export function validarTexto(texto, minimo = 1, maximo = 100) {
  if (typeof texto !== "string") return false;
  const limpio = texto.trim();
  return limpio.length >= minimo && limpio.length <= maximo;
}

/* validación correo */
export function validarCorreo(correo) {
  if (typeof correo !== "string") return false;
  const limpio = correo.trim().toLowerCase();
  if (limpio.length === 0 || limpio.length > 100) return false;

  const regexFormato = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!regexFormato.test(limpio)) return false;

  return (
    limpio.endsWith("@duoc.cl") ||
    limpio.endsWith("@profesor.duoc.cl") ||
    limpio.endsWith("@gmail.com")
  );
}

/* validación run */
export function validarRun(run) {
  if (typeof run !== "string") return false;
  const limpio = run.trim();
  if (limpio.length < 7 || limpio.length > 9) return false;
  if (limpio.includes(".") || limpio.includes("-")) return false;

  // formato: números seguidos opcionalmente de 'k' o 'k'
  const regexRun = /^[0-9]+[0-9kK]$/;
  return regexRun.test(limpio);
}

/* validación contraseña */
export function validarPassword(password, min = 4, max = 10) {
  if (typeof password !== "string") return false;
  const limpia = password.trim();
  return limpia.length >= min && limpia.length <= max;
}

/* validación números */
export function validarNumeroPositivo(valor, permitirDecimal = false) {
  const num = Number(valor);
  if (isNaN(num) || num < 0) return false;
  if (!permitirDecimal && !Number.isInteger(num)) return false;
  return true;
}

import { Navigate } from "react-router-dom";
import { estaAutenticado, esAdmin } from "../auth/auth.js";

/**
 * componente guardián de rutas
 * si no hay sesión activa, redirige automáticamente a /login con el replace
 * si la ruta exige rol de admin y el usuario no lo tiene, redirige a main /.
 */
export default function RutaProtegida({ children, exigirAdmin = true }) {
  if (!estaAutenticado()) {
    return <Navigate to="/login" replace />;
  }

  if (exigirAdmin && !esAdmin()) {
    alert("Acceso denegado: se requieren permisos de Administrador.");
    return <Navigate to="/" replace />;
  }

  return children;
}

function CarritoLateral({
  carrito,
  vaciarCarrito,
  comprar,
  agregarAlCarrito,
  disminuirCantidad,
  eliminarDelCarrito,
}) {
  const totalPagar = calcularTotal(carrito);

  return (
    <aside>
      <h2>Carrito</h2>
      <div className="col-4">
        <div id="cart-list">
          {carrito.length === 0 ? (
            <p>El carrito está vacío.</p>
          ) : (
            carrito.map((item) => (
              <div key={item.id} className="cart-item">
                <strong>{item.nombre}</strong>
                <br />
                Precio: ${item.precio} x {item.cantidad} = ${item.precio * item.cantidad}

                <button onClick={() => disminuirCantidad(item.id)}>-</button>
                <button onClick={() => eliminarDelCarrito(item.id)}>

                </button>
              </div>
            ))
          )}
        </div>

        {carrito.length > 0 && <h3>Total: ${totalPagar}</h3>}

        <button
          id="vaciar-carrito"
          className="btn-secundario"
          onClick={vaciarCarrito}
        >
          Vaciar Carrito
        </button>
      </div>
      <button id="Comprar" onClick={comprar}>

import { useState } from "react";
import { NavLink, Link, useNavigate } from "react-router-dom";
import { cerrarSesion } from "../auth/auth.js";

function AdminHome({
  productos,
  setProductos,
  setPaginaActual,
}) {
  // formulario
  const [formData, setFormData] = useState({
    codigo: "",
    nombre: "",
    categoria: "",
    precio: "",
    stock: "",
    unidad: "",
    descripcion: "",
  });

  const navigate = useNavigate();

  const handleLogout = () => {
    cerrarSesion();
    navigate("/login");
  };

  // errores y mensajes de éxito
  const [errores, setErrores] = useState([]);
  const [mensajeExito, setMensajeExito] = useState(null);

  // cambios en los inputs
  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  // cargar datos de un producto existente para su edición
  const cargarParaEditar = (producto) => {
    setFormData({
      codigo: producto.id,
      nombre: producto.nombre,
      categoria: producto.categoria || "",
      precio: producto.precio.toString(),
      stock: (producto.stock ?? 100).toString(),
      unidad: producto.unidad || "Kilo",
      descripcion: producto.descripcion || "",
    });
    setErrores([]);
    setMensajeExito(null);
    window.scrollTo({ top: 0, behavior: "smooth" });
  };

  // eliminar un producto del catálogo
  const handleEliminar = (id) => {
    if (window.confirm(`¿Estás seguro de eliminar el producto con ID ${id}?`)) {
      setProductos((actuales) => actuales.filter((p) => p.id !== id));
      setMensajeExito({
        texto: `Producto ${id} eliminado correctamente del catálogo.`,
        datos: null,
      });
    }
  };

  // validaciones de nuevoproducto.js
  const handleSubmit = (e) => {
    e.preventDefault();

    const codigo = formData.codigo.trim();
    const nombre = formData.nombre.trim();
    const categoria = formData.categoria;
    const precioTexto = formData.precio.trim();
    const stockTexto = formData.stock.trim();
    const unidad = formData.unidad;
    const descripcion = formData.descripcion.trim();

    const precio = Number(precioTexto);
    const stock = Number(stockTexto);

    const nuevosErrores = [];

    // id / código
    if (codigo === "") {
      nuevosErrores.push("El código (ID) del producto es obligatorio.");
    } else if (codigo.length < 3) {
      nuevosErrores.push("El código debe tener al menos 3 caracteres.");
    }

    // nombre
    if (nombre === "") {
      nuevosErrores.push("El nombre del producto es obligatorio.");
    } else if (nombre.length > 100) {
      nuevosErrores.push("El nombre no puede superar los 100 caracteres.");
    }

    // categoría
    if (categoria === "") {
      nuevosErrores.push("Debe seleccionar una categoría.");
    }

    // validar precio
    if (precioTexto === "") {
      nuevosErrores.push("El precio es obligatorio.");
    } else if (isNaN(precio) || precio < 0) {
      nuevosErrores.push("El precio debe ser un número válido mayor o igual a 0.");
    }

    // validar stock
    if (stockTexto === "") {
      nuevosErrores.push("El stock es obligatorio.");
    } else if (isNaN(stock) || stock < 0 || !Number.isInteger(stock)) {
      nuevosErrores.push("El stock debe ser un número entero mayor o igual a 0.");
    }

    // validar unidad de medida
    if (unidad === "") {
      nuevosErrores.push("Debe seleccionar una unidad de medida.");
    }

    // validar descripción
    if (descripcion.length > 500) {
      nuevosErrores.push("La descripción no puede superar los 500 caracteres.");
    }

    if (nuevosErrores.length > 0) {
      setErrores(nuevosErrores);
      setMensajeExito(null);
    } else {
      setErrores([]);

      const nuevoProducto = {
        id: codigo,
        categoria: categoria,
        nombre: nombre,
        descripcion: descripcion,
        precio: precio,
        stock: stock,
        unidad: unidad,
      };

      // si id existe, se edita. si no, se agrega
      setProductos((actuales) => {
        const existe = actuales.some((p) => p.id === codigo);
        if (existe) {
          return actuales.map((p) => (p.id === codigo ? nuevoProducto : p));
        } else {
          return [...actuales, nuevoProducto];
        }
      });

      setMensajeExito({
        texto: "✅ Producto guardado correctamente en el catálogo.",
        datos: nuevoProducto,
      });

      // limpiar formulario
      setFormData({
        codigo: "",
        nombre: "",
        categoria: "",
        precio: "",
        stock: "",
        unidad: "",
        descripcion: "",
      });
    }
  };

  return (
    <div className="admin-page">
      {/* navegación del admin */}
      <header className="admin-header">
        <h2>Panel Administrador</h2>
        <nav>
          <NavLink to="/admin" end>Cambiar Productos</NavLink>
          <NavLink to="/admin/usuarios">Cambiar Usuarios</NavLink>
          <Link to="/" className="btn-secundario">Volver a la Tienda</Link>
          <button
            type="button"
            className="btn-secundario"
            onClick={handleLogout}
            style={{ backgroundColor: "#E74C3C", color: "#fff", cursor: "pointer" }}
          >
            Cerrar Sesión
          </button>
        </nav>
      </header>

      <main style={{ gridTemplateColumns: "1fr", padding: "1.5rem" }}>
        <h1>Hola Admin</h1>

        <div className="admin-container">
          <h3>
            Añadir Producto / Editar producto (Ingresar mismo ID para editar un
            producto existente)
          </h3>

          <form id="form-producto" onSubmit={handleSubmit}>
            <div className="campo">
              <label htmlFor="codigo">Código (ID)</label>
              <input
                type="text"
                id="codigo"
                name="codigo"
                placeholder="Ej: FR001"
                value={formData.codigo}
                onChange={handleChange}
              />
            </div>

            <div className="campo">
              <label htmlFor="nombre">Nombre del Producto</label>
              <input
                type="text"
                id="nombre"
                name="nombre"

  carrito,

  comprar,

  disminuirCantidad,
  eliminarDelCarrito,
  setPaginaActual,
  setUsuarioSesion,
}) {

  // estados para los inputs
  const [usuario, setUsuario] = useState("");
  const [password, setPassword] = useState("");

  // estados para los mensajes de error y estado de login
  const [errores, setErrores] = useState([]);
  const [esAdmin, setEsAdmin] = useState(false);
  const [esCliente, setEsCliente] = useState(false);

    reingresoPassword: "",
    telefono: "",
    region: "",
    direccion: "",
  });

  // estados para errores y registro
  const [errores, setErrores] = useState([]);
  const [registroExitoso, setRegistroExitoso] = useState(null);

  // manejador para actualizar cualquier input
  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  // función envío
  const handleSubmit = (e) => {
    e.preventDefault();

    const runLimpio = formData.run.trim();
    const nombreLimpio = formData.nombre.trim();
    const apellidosLimpio = formData.apellidos.trim();
    const correoLimpio = formData.correo.trim();
    const reingresoCorreoLimpio = formData.reingresoCorreo.trim();
    const passwordLimpia = formData.password.trim();
    const reingresoPasswordLimpia = formData.reingresoPassword.trim();
    const telefonoLimpio = formData.telefono.trim();
    const regionSeleccionada = formData.region;
    const direccionLimpia = formData.direccion.trim();

    const nuevosErrores = [];

    // validar run
    if (!validarRun(runLimpio)) {
      nuevosErrores.push("El RUN debe ser sin puntos ni guion y tener entre 7 y 9 caracteres (Ej: 19011022K).");
    }

    // validar nombre
    if (!validarTexto(nombreLimpio, 1, 50)) {
      nuevosErrores.push("El nombre es obligatorio y no puede superar los 50 caracteres.");
    }

    // validar apellidos
    if (!validarTexto(apellidosLimpio, 1, 100)) {
      nuevosErrores.push("Los apellidos son obligatorios y no pueden superar los 100 caracteres.");
    }

    // validar correo
    if (!validarCorreo(correoLimpio)) {
      nuevosErrores.push("El correo debe ser @duoc.cl, @profesor.duoc.cl o @gmail.com.");
    }

    // validar reingreso de correo
    if (reingresoCorreoLimpio === "") {
      nuevosErrores.push("Debe reingresar su correo.");
    } else if (correoLimpio !== reingresoCorreoLimpio) {
      nuevosErrores.push("Los correos ingresados no coinciden.");
    }

    // validar contraseña con función pura
    if (!validarPassword(passwordLimpia, 4, 10)) {
      nuevosErrores.push("La contraseña debe tener entre 4 y 10 caracteres.");
    }

    // validar reingreso de contraseña
    if (reingresoPasswordLimpia === "") {
      nuevosErrores.push("Debe reingresar su contraseña.");
    } else if (passwordLimpia !== reingresoPasswordLimpia) {
      nuevosErrores.push("Las contraseñas ingresadas no coinciden.");
    }

    // validar región
    if (regionSeleccionada === "") {
      nuevosErrores.push("Debe seleccionar una región.");
    }

    // validar dirección
    if (!validarTexto(direccionLimpia, 1, 300)) {
      nuevosErrores.push("La dirección es obligatoria y no puede superar los 300 caracteres.");
    }

    // procesar resultados
    if (nuevosErrores.length > 0) {
      setErrores(nuevosErrores);
      setRegistroExitoso(null);
    } else {
      setErrores([]);
      const nuevoUsuario = {
        run: runLimpio,
        nombre: nombreLimpio,
        apellidos: apellidosLimpio,
        correo: correoLimpio,
        password: passwordLimpia,
        telefono: telefonoLimpio,
        region: regionSeleccionada,
        direccion: direccionLimpia,
      };

      // guardado en localstorage 
      const usuariosRegistrados =
        JSON.parse(localStorage.getItem("usuariosRegistrados")) || [];
      usuariosRegistrados.push(nuevoUsuario);
      localStorage.setItem(
        "usuariosRegistrados",
        JSON.stringify(usuariosRegistrados)
      );

      setRegistroExitoso(nuevoUsuario);

      // limpiar el formulario
      setFormData({
        run: "",
        nombre: "",
        apellidos: "",
        correo: "",
        reingresoCorreo: "",
        password: "",
        reingresoPassword: "",
        telefono: "",
        region: "",
        direccion: "",
      });
    }
  };

  return (
    <main>
      <div className="contenido-principal">
        <form id="registro" onSubmit={handleSubmit}>
          <h2>Registro</h2>

          {/* run */}
          <div className="campo">
            <label htmlFor="run">RUN / RUT (sin puntos ni guion)</label>
            <input
              type="text"
              id="run"
              name="run"
              maxLength="9"
              placeholder="Ej: 19011022K"
              value={formData.run}
              onChange={handleChange}
            />
          </div>

          {/* nombre */}
          <div className="campo">
            <label htmlFor="nombre">Nombre</label>
            <input
              type="text"
              id="nombre"
              name="nombre"
              maxLength="50"
              placeholder="Ej: Ana"
              value={formData.nombre}
              onChange={handleChange}
            />
          </div>

          {/* apellidos */}
          <div className="campo">
            <label htmlFor="apellidos">Apellidos</label>
            <input
              type="text"
              id="apellidos"
              name="apellidos"
              maxLength="100"
              placeholder="Ej: Silva Pérez"
              value={formData.apellidos}
              onChange={handleChange}
            />
          </div>

          {/* correo */}
          <div className="campo">
            <label htmlFor="correo">Correo</label>
            <input
              type="email"
              id="correo"
              name="correo"
              maxLength="100"
              placeholder="Ej: ana@duoc.cl"
              value={formData.correo}
              onChange={handleChange}
            />
          </div>

          {/* reingreso correo */}
          <div className="campo">
            <label htmlFor="reingreso-correo">Reingrese Correo</label>
            <input
              type="email"
              id="reingreso-correo"
              name="reingresoCorreo"
              placeholder="Ej: ana@duoc.cl"
              value={formData.reingresoCorreo}
              onChange={handleChange}
            />
          </div>

          {/* contraseña */}
          <div className="campo">
            <label htmlFor="password">Contraseña</label>
            <input
              type="password"
              id="password"
              name="password"
              placeholder="Ej: 123456"
              value={formData.password}
              onChange={handleChange}
            />
          </div>

          {/* reingreso contraseña */}
          <div className="campo">
            <label htmlFor="reingreso-password">Reingrese Contraseña</label>
            <input
              type="password"
              id="reingreso-password"
              name="reingresoPassword"
              placeholder="Ej: 123456"
              value={formData.reingresoPassword}
              onChange={handleChange}
            />
          </div>

          {/* teléfono */}
          <div className="campo">
            <label htmlFor="telefono">Teléfono (Opcional)</label>
            <input
              type="number"
              id="telefono"
              name="telefono"
              placeholder="Ej: 987654321"
              value={formData.telefono}
              onChange={handleChange}
            />
          </div>

          {/* región */}
          <div className="campo">
            <label htmlFor="region">Región</label>
            <select
              id="region"
              name="region"
              value={formData.region}
              onChange={handleChange}
            >
              <option value="">Seleccione una región</option>
              <option value="arica">Región de Arica y Parinacota</option>
              <option value="tarapaca">Región de Tarapacá</option>
              <option value="antofagasta">Región de Antofagasta</option>
              <option value="atacama">Región de Atacama</option>
              <option value="coquimbo">Región de Coquimbo</option>
              <option value="valparaiso">Región de Valparaíso</option>
              <option value="rm">Región Metropolitana de Santiago</option>
              <option value="ohiggins">
                Región del Libertador General Bernardo O'Higgins
              </option>
              <option value="maule">Región del Maule</option>
              <option value="nuble">Región de Ñuble</option>
              <option value="biobio">Región del Biobío</option>
              <option value="araucania">Región de La Araucanía</option>
              <option value="los_rios">Región de Los Ríos</option>
              <option value="los_lagos">Región de Los Lagos</option>
              <option value="aysen">
                Región Aysén del General Carlos Ibáñez del Campo
              </option>
              <option value="magallanes">
                Región de Magallanes y de la Antártica Chilena
              </option>
            </select>
          </div>

          {/* dirección */}
          <div className="campo">
            <label htmlFor="direccion">Dirección de Entrega</label>
            <input
              type="text"
              id="direccion"
              name="direccion"
              maxLength="300"
              placeholder="Ej: Av. Las Flores 1234"
              value={formData.direccion}
              onChange={handleChange}
            />
          </div>

          <button type="submit" className="btn-principal">
            Registrarse
          </button>
        </form>

        {/* zona para mensajes */}
        <div id="mensajes">
          {errores.length > 0 &&
            errores.map((error, index) => (
              <div key={index} className="error">
                ⚠️ {error}
              </div>
            ))}

          {registroExitoso && (
            <div className="exito">
              ✅ Registro realizado correctamente.
              <br />
              <br />
              <strong>RUN:</strong> {registroExitoso.run}
              <br />
              <strong>Nombre:</strong> {registroExitoso.nombre}{" "}
              {registroExitoso.apellidos}
              <br />
              <strong>Correo:</strong> {registroExitoso.correo}
              <br />
              <strong>Región:</strong> {registroExitoso.region}
              <br />
              <strong>Dirección:</strong> {registroExitoso.direccion}
              {registroExitoso.telefono && (
                <>
                  <br />
                  <strong>Teléfono:</strong> {registroExitoso.telefono}
                </>
              )}
            </div>
          )}
        </div>
      </div>

      {/* carrito lateral */}
      <CarritoLateral
        carrito={carrito}
        vaciarCarrito={vaciarCarrito}
        comprar={comprar}
        agregarAlCarrito={agregarAlCarrito}
        disminuirCantidad={disminuirCantidad}
        eliminarDelCarrito={eliminarDelCarrito}
      />
    </main>
  );
}

export default Registro;
