import { useState, useEffect } from "react";
import { Routes, Route } from "react-router-dom";
import LayoutPrincipal from "./layouts/LayoutPrincipal";
import Home from "./pages/Home";
import Productos from "./pages/Productos";
import Contacto from "./pages/Contacto";
import Login from "./pages/Login";
import Registro from "./pages/Registro";
import Blog from "./pages/Blog";
import AdminHome from "./pages/AdminHome";
import AdminUsuarios from "./pages/AdminUsuarios";
import RutaProtegida from "./components/RutaProtegida";
import { calcularCantidadTotal } from "./utils/carrito.js";
import { getProductos } from "./services/productoService.js";

function App() {
  const [carrito, setCarrito] = useState(() => {
    const guardado = localStorage.getItem("carrito");
    return guardado ? JSON.parse(guardado) : [];
  });

  useEffect(() => {
    localStorage.setItem("carrito", JSON.stringify(carrito));
  }, [carrito]);

  const [productos, setProductos] = useState([]);

  const cargarProductos = async () => {
    try {
      const data = await getProductos();
      setProductos(data);
    } catch (error) {
      console.error("Error al obtener productos de Spring Boot:", error);
    }
  };

  useEffect(() => {
    cargarProductos();
  }, []);

  function agregarAlCarrito(id) {
    setCarrito((carritoActual) => {
      const itemExistente = carritoActual.find((item) => item.id === id || item.codigo === id);
      if (itemExistente) {
        if (itemExistente.cantidad >= 5) {
          alert("Máximo 5 unidades por producto");
          return carritoActual;
        }
        return carritoActual.map((item) =>
          (item.id === id || item.codigo === id) ? { ...item, cantidad: item.cantidad + 1 } : item
        );
      } else {
        const prod = productos.find((p) => p.id === id || p.codigo === id);
        if (!prod) return carritoActual;
        return [...carritoActual, { ...prod, cantidad: 1 }];
      }
    });
  }

  function disminuirCantidad(id) {
    setCarrito((carritoActual) => {
      const item = carritoActual.find((i) => i.id === id || i.codigo === id);
      if (!item) return carritoActual;
      if (item.cantidad <= 1) {
        return carritoActual.filter((i) => i.id !== id && i.codigo !== id);
      } else {
        return carritoActual.map((i) =>
          (i.id === id || i.codigo === id) ? { ...i, cantidad: item.cantidad - 1 } : i
        );
      }
    });
  }

  function eliminarDelCarrito(id) {
    setCarrito((carritoActual) => carritoActual.filter((item) => item.id !== id && item.codigo !== id));
  }

  function vaciarCarrito() {
    setCarrito([]);
  }

  function comprar() {
    if (carrito.length === 0) {
      alert("El carrito está vacío.");
      return;
    }
    alert("¡Gracias por tu compra en HuertoHogar!");
    vaciarCarrito();
  }

  const cantidadTotalCarrito = calcularCantidadTotal(carrito);

  const propsComunes = {
    carrito,
    vaciarCarrito,
    comprar,
    agregarAlCarrito,
    disminuirCantidad,
    eliminarDelCarrito,
    productos,
  };

  return (
    <Routes>
      <Route element={<LayoutPrincipal cantidadCarrito={cantidadTotalCarrito} />}>
        <Route index element={<Home {...propsComunes} />} />
        <Route path="productos" element={<Productos {...propsComunes} />} />
        <Route path="contacto" element={<Contacto {...propsComunes} />} />
        <Route path="login" element={<Login {...propsComunes} />} />
        <Route path="registro" element={<Registro {...propsComunes} />} />
        <Route path="blog" element={<Blog {...propsComunes} />} />
      </Route>

      <Route
        path="admin"
        element={
          <RutaProtegida exigirAdmin={true}>
            <AdminHome productos={productos} setProductos={setProductos} recargarProductos={cargarProductos} />
          </RutaProtegida>
        }
      />
      <Route
        path="admin/usuarios"
        element={
          <RutaProtegida exigirAdmin={true}>
            <AdminUsuarios />
          </RutaProtegida>
        }
      />
    </Routes>
  );
}

export default App;

import React, { createContext, useContext, useState, useEffect, useCallback } from 'react';
import { request } from '../services/api.js';

const AuthContext = createContext(null);
const TOKEN_KEY = 'hh_jwt_token';

export function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    const [token, setToken] = useState(() => sessionStorage.getItem(TOKEN_KEY) || null);
    const [loading, setLoading] = useState(true);

    const checkAuth = useCallback(async () => {
        const savedToken = sessionStorage.getItem(TOKEN_KEY);
        if (!savedToken) {
            setUser(null);
            setToken(null);
            setLoading(false);
            return;
        }

        try {
            const data = await request('/auth/me', { token: savedToken });
            setUser({ username: data.username, rol: data.rol });
            setToken(savedToken);
        } catch (error) {
            sessionStorage.removeItem(TOKEN_KEY);
            setUser(null);
            setToken(null);
        } finally {
            setLoading(false);
        }
    }, []);

    useEffect(() => {
        checkAuth();
    }, [checkAuth]);

    const login = async (username, password) => {
        const data = await request('/auth/login', {
            method: 'POST',
            body: { username, password },
        });

        sessionStorage.setItem(TOKEN_KEY, data.token);
        setToken(data.token);
        setUser({ username: data.username, rol: data.rol });
        return data;
    };

    const logout = () => {
        sessionStorage.removeItem(TOKEN_KEY);
        setToken(null);
        setUser(null);
    };

    const esAdmin = user?.rol === 'ADMIN' || user?.rol === 'ROLE_ADMIN';
    const usuario = user;
    const cargando = loading;

    return (
        <AuthContext.Provider value={{ user, usuario, token, login, logout, loading, cargando, esAdmin, checkAuth }}>
            {children}
        </AuthContext.Provider>
    );
}

export function useAuth() {
    const context = useContext(AuthContext);
    if (!context) {
        throw new Error('useAuth debe usarse dentro de un AuthProvider');
    }
    return context;
}

const CLAVE_SESION = "usuarioActual";

/**
 * valida credenciales e inicia sesión.
 * soporta las credenciales oficiales de huertohogar (admin@duoc.cl / admin)
 * y las didácticas del tutorial docente (admin / admin123).
 */
export function iniciarSesion(usuario, password) {
  const usuarioLimpio = usuario.trim();
  const passwordLimpia = password.trim();

  // caso 1: administrador
  if (
    (usuarioLimpio === "admin@duoc.cl" && passwordLimpia === "admin") ||
    (usuarioLimpio === "admin" && passwordLimpia === "admin123")
  ) {
    const sesion = { usuario: usuarioLimpio, rol: "ADMIN" };
    localStorage.setItem(CLAVE_SESION, JSON.stringify(sesion));
    return { exito: true, rol: "ADMIN", sesion };
  }

  // caso 2: cliente general válido (según reglas de dominio institucional o gmail)
  const esCorreoValido =
    usuarioLimpio.endsWith("@duoc.cl") ||
    usuarioLimpio.endsWith("@profesor.duoc.cl") ||
    usuarioLimpio.endsWith("@gmail.com");

  if (esCorreoValido && passwordLimpia.length >= 4 && passwordLimpia.length <= 10) {
    const sesion = { usuario: usuarioLimpio, rol: "CLIENTE" };
    localStorage.setItem(CLAVE_SESION, JSON.stringify(sesion));
    return { exito: true, rol: "CLIENTE", sesion };
  }

  return { exito: false, error: "Credenciales incorrectas." };
}

/**
 * obtiene el objeto de sesión almacenado en localstorage.
 */
export function obtenerSesion() {
  try {
    const datos = localStorage.getItem(CLAVE_SESION);
    return datos ? JSON.parse(datos) : null;
  } catch (error) {
    console.error("Error al leer sesión:", error);
    return null;
  }
}

/**
 * elimina la sesión activa de localstorage.
 */
export function cerrarSesion() {
  localStorage.removeItem(CLAVE_SESION);
}

/**
 * revisa si existe un usuario autenticado
 */
export function estaAutenticado() {
  return obtenerSesion() !== null;
}

/**
 * revisa si el usuario autenticado tiene rol de admin
 */
export function esAdmin() {
  const sesion = obtenerSesion();
  return sesion?.rol === "ADMIN";
}

/* cálculo del monto total a pagar de los productos en el carrito */
export function calcularTotal(carrito) {
  if (!Array.isArray(carrito)) return 0;
  return carrito.reduce(
    (total, item) => total + (item.precio || 0) * (item.cantidad || 0),
    0
  );
}

/* cálculo de la cantidad total de unidades en el carrito. */
export function calcularCantidadTotal(carrito) {
  if (!Array.isArray(carrito)) return 0;
  return carrito.reduce(
    (total, item) => total + (item.cantidad || 0),
    0
  );
}

/* validación de un texto según longitud dsps de eliminar espacios laterales. */
export function validarTexto(texto, minimo = 1, maximo = 100) {
  if (typeof texto !== "string") return false;
  const limpio = texto.trim();
  return limpio.length >= minimo && limpio.length <= maximo;
}

/* validación correo */
export function validarCorreo(correo) {
  if (typeof correo !== "string") return false;
  const limpio = correo.trim().toLowerCase();
  if (limpio.length === 0 || limpio.length > 100) return false;

  const regexFormato = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!regexFormato.test(limpio)) return false;

  return (
    limpio.endsWith("@duoc.cl") ||
    limpio.endsWith("@profesor.duoc.cl") ||
    limpio.endsWith("@gmail.com")
  );
}

/* validación run */
export function validarRun(run) {
  if (typeof run !== "string") return false;
  const limpio = run.trim();
  if (limpio.length < 7 || limpio.length > 9) return false;
  if (limpio.includes(".") || limpio.includes("-")) return false;

  // formato: números seguidos opcionalmente de 'k' o 'k'
  const regexRun = /^[0-9]+[0-9kK]$/;
  return regexRun.test(limpio);
}

/* validación contraseña */
export function validarPassword(password, min = 4, max = 10) {
  if (typeof password !== "string") return false;
  const limpia = password.trim();
  return limpia.length >= min && limpia.length <= max;
}

/* validación números */
export function validarNumeroPositivo(valor, permitirDecimal = false) {
  const num = Number(valor);
  if (isNaN(num) || num < 0) return false;
  if (!permitirDecimal && !Number.isInteger(num)) return false;
  return true;
}

import React from 'react';
import { Navigate, useLocation } from 'react-router-dom';
import { useAuth } from '../context/AuthContext.jsx';

export default function RutaProtegida({ children, exigirAdmin = false }) {
    const { user, usuario, loading, cargando, esAdmin } = useAuth();
    const location = useLocation();

    const estaCargando = loading || cargando;
    const usuarioActivo = user || usuario;

    if (estaCargando) {
        return (
            <div style={{ textAlign: "center", padding: "3rem" }}>
                <p>Verificando sesión con el servidor...</p>
            </div>
        );
    }

    if (!usuarioActivo) {
        return <Navigate to="/login" state={{ from: location.pathname }} replace />;
    }

    const esAdministrador = esAdmin || usuarioActivo.rol === 'ADMIN' || usuarioActivo.rol === 'ROLE_ADMIN';

    if (exigirAdmin && !esAdministrador) {
        alert("Acceso denegado: se requieren permisos de Administrador.");
        return <Navigate to="/" replace />;
    }

    return children;
}

function CarritoLateral({

  vaciarCarrito,
  comprar,
  agregarAlCarrito,
  disminuirCantidad,
  eliminarDelCarrito,
}) {
  const totalPagar = calcularTotal(carrito);

  return (
    <aside>
      <h2>Carrito</h2>
      <div className="col-4">
        <div id="cart-list">
          {carrito.length === 0 ? (

                <br />

import React, { useEffect, useState, useCallback } from 'react';
import { useAuth } from '../context/AuthContext.jsx';
import { getProductos, crearProducto, actualizarProducto, eliminarProducto } from '../services/productoService.js';

const FORM_INICIAL = {
    codigo: '',
    nombre: '',
    categoria: 'Frutas',
    precio: 0,
    stock: 0,
    unidad: 'Kilo',
    descripcion: ''
};

export default function AdminHome() {
    const { token, logout } = useAuth();
    const [productos, setProductos] = useState([]);
    const [form, setForm] = useState(FORM_INICIAL);
    const [editandoId, setEditandoId] = useState(null);
    const [cargando, setCargando] = useState(true);
    const [mensaje, setMensaje] = useState({ tipo: '', texto: '' });

    const cargarProductos = useCallback(async () => {
        setCargando(true);
        try {
            const data = await getProductos();
            setProductos(data || []);
        } catch (err) {
            setMensaje({ tipo: 'danger', texto: err.message || 'Error al cargar los productos' });
        } finally {
            setCargando(false);
        }
    }, []);

    useEffect(() => {
        cargarProductos();
    }, [cargarProductos]);

    const handleChange = (e) => {
        const { name, value } = e.target;
        setForm(prev => ({
            ...prev,
            [name]: name === 'precio' || name === 'stock' ? Number(value) : value
        }));
    };

    const handleSubmit = async (e) => {
        e.preventDefault();
        setMensaje({ tipo: '', texto: '' });

        try {
            if (editandoId) {
                await actualizarProducto(editandoId, form, token);
                setMensaje({ tipo: 'success', texto: 'Producto actualizado con éxito en Spring Boot.' });
            } else {
                await crearProducto(form, token);
                setMensaje({ tipo: 'success', texto: 'Producto registrado exitosamente en la Base de Datos.' });
            }
            setForm(FORM_INICIAL);
            setEditandoId(null);
            cargarProductos();
        } catch (err) {
            if (err.status === 401 || err.status === 403) {
                logout();
            } else {
                setMensaje({ tipo: 'danger', texto: err.message || 'Error al procesar la solicitud' });
            }
        }
    };

    const handleEditar = (prod) => {
        setEditandoId(prod.id);
        setForm({
            codigo: prod.codigo || '',
            nombre: prod.nombre || '',
            categoria: prod.categoria || 'Frutas',
            precio: prod.precio || 0,
            stock: prod.stock || 0,
            unidad: prod.unidad || 'Kilo',
            descripcion: prod.descripcion || ''
        });
        setMensaje({ tipo: 'info', texto: `Editando producto ID: ${prod.id}` });
    };

    const handleEliminar = async (id) => {
        if (!window.confirm('¿Estás seguro de eliminar este producto del catálogo?')) return;
        setMensaje({ tipo: '', texto: '' });

        try {
            await eliminarProducto(id, token);
            setMensaje({ tipo: 'success', texto: 'Producto eliminado exitosamente.' });
            cargarProductos();
        } catch (err) {
            if (err.status === 401 || err.status === 403) {
                logout();
            } else {
                setMensaje({ tipo: 'danger', texto: err.message || 'Error al eliminar el producto' });
            }
        }
    };

    const handleCancelar = () => {
        setForm(FORM_INICIAL);
        setEditandoId(null);
        setMensaje({ tipo: '', texto: '' });
    };

    return (
        <main className="container my-5">
            <h1 className="mb-4">Mantenedor de Productos (Administrador)</h1>

            {mensaje.texto && (
                <div className={`alert alert-${mensaje.tipo} alert-dismissible fade show`} role="alert">
                    {mensaje.texto}
                    <button type="button" className="btn-close" onClick={() => setMensaje({ tipo: '', texto: '' })}></button>
                </div>
            )}

            <div className="row g-4">
                {/* Formulario Crear / Editar */}
                <div className="col-lg-5">
                    <div className="card shadow-sm">
                        <div className="card-header bg-success text-white">
                            <h5 className="card-title mb-0">
                                {editandoId ? 'Editar Producto' : 'Nuevo Producto'}
                            </h5>
                        </div>
                        <div className="card-body">
                            <form onSubmit={handleSubmit}>
                                <div className="mb-3">
                                    <label className="form-label">Código</label>
                                    <input type="text" name="codigo" className="form-control" value={form.codigo} onChange={handleChange} required placeholder="Ej: FR001" />
                                </div>
                                <div className="mb-3">
                                    <label className="form-label">Nombre</label>
                                    <input type="text" name="nombre" className="form-control" value={form.nombre} onChange={handleChange} required placeholder="Ej: Manzanas Fuji" />

import { validarTexto, validarCorreo } from "../utils/validaciones.js";

function Contacto({
  carrito,
  vaciarCarrito,
  comprar,
  agregarAlCarrito,
  disminuirCantidad,
  eliminarDelCarrito,
}) {
  // estados para los campos del formulario
  const [nombre, setNombre] = useState("");
  const [correo, setCorreo] = useState("");
  const [comentario, setComentario] = useState("");

  // estados para los mensajes de validación y confirmación
  const [errores, setErrores] = useState([]);
  const [exito, setExito] = useState("");

  // función de validación y envío
  const handleSubmit = (e) => {
    e.preventDefault(); // evita la recarga automática del navegador

    const nombreLimpio = nombre.trim();
    const correoLimpio = correo.trim();
    const comentarioLimpio = comentario.trim();
    const listaErrores = [];

    // validaciones de nombre con función pura
    if (!validarTexto(nombreLimpio, 1, 100)) {
      listaErrores.push("El nombre es obligatorio y no puede superar los 100 caracteres.");
    }

    // validaciones de correo con función pura
    if (!validarCorreo(correoLimpio)) {
      listaErrores.push("El correo debe ser @duoc.cl, @profesor.duoc.cl o @gmail.com.");
    }

    // validaciones de comentario con función pura
    if (!validarTexto(comentarioLimpio, 1, 500)) {
      listaErrores.push("El mensaje es obligatorio y no puede superar los 500 caracteres.");
    }

    // actualización de estado según resultado
    if (listaErrores.length > 0) {
      setErrores(listaErrores);
      setExito("");
    } else {
      setErrores([]);
      setExito(`Gracias por contactarte con HuertoHogar, ${nombreLimpio}.`);
      // limpieza de campos controlados
      setNombre("");
      setCorreo("");
      setComentario("");
    }
  };

  return (
    <main>
      <div className="contenido-principal">
        <form id="form-contacto" onSubmit={handleSubmit}>
          <h2>Formulario de Contacto</h2>

          {/* campo nombre */}
          <div className="campo">
            <label htmlFor="nombre">Nombre Completo</label>
            <input
              type="text"
              id="nombre"
              name="nombre"
              maxLength="100"
              placeholder="Ej: Ana Pérez"
              value={nombre}
              onChange={(e) => setNombre(e.target.value)}
            />
          </div>

          {/* campo correo */}
          <div className="campo">
            <label htmlFor="correo">Correo Electrónico</label>
            <input
              type="email"
              id="correo"
              name="correo"
              maxLength="100"
              placeholder="Ej: ana@duoc.cl"
              value={correo}
              onChange={(e) => setCorreo(e.target.value)}
            />
          </div>

          {/* campo mensaje / comentario */}
          <div className="campo">
            <label htmlFor="comentario">Mensaje o Comentario</label>
            <textarea
              id="comentario"
              name="comentario"
              rows="5"
              maxLength="500"
              placeholder="Escriba su comentario"
              value={comentario}
              onChange={(e) => setComentario(e.target.value)}
            />
          </div>

          <button type="submit" className="btn-principal">
            Enviar
          </button>

          {/* contenedor de alertas dinámicas */}
          <div id="mensajes">
            {errores.length > 0 &&
              errores.map((error, index) => (
                <div key={index} className="error">
                  ⚠️ {error}
                </div>
              ))}

            {exito && (

const API_BASE_URL = 'http://localhost:8080';

export async function request(endpoint, options = {}) {
    const { token, body, headers = {}, method = 'GET', ...customConfig } = options;

    const config = {
        method,
        headers: { ...headers },
        ...customConfig,
    };

    if (body) {
        config.headers['Content-Type'] = 'application/json';
        config.body = typeof body === 'string' ? body : JSON.stringify(body);
    }

    if (token) {
        config.headers['Authorization'] = `Bearer ${token}`;
    }

    try {
        const response = await fetch(`${API_BASE_URL}${endpoint}`, config);
        if (response.status === 204) return null;

        const data = await response.json().catch(() => null);

        if (!response.ok) {
            const errorMessage = data?.message || `Error HTTP ${response.status}`;
            const error = new Error(errorMessage);
            error.status = response.status;
            throw error;
        }

        return data;
    } catch (error) {
        if (!error.status) {
            error.message = 'No se pudo conectar con el servidor Spring Boot (http://localhost:8080).';
        }
        throw error;
    }
}

# Plantilla de backend para proyecto EP3 — DSY1104

**Es intencionalmente una plantilla, no una solución de la evaluación.** Incluye JDK 21, Spring Boot 4.1.1, Maven Wrapper, H2, Swagger, validación y un endpoint técnico.

## Inicio
1. Abrir la carpeta en IntelliJ/VS Code.
1.1 : darle permisos con chmod +x mvnw
2. Ejecutar `./mvnw.cmd spring-boot:run` en PowerShell o `mvnw.cmd spring-boot:run` en CMD.
o ./mvnw spring-boot:run en macos
3. Abrir `http://localhost:8080/api/salud` (respuesta de prueba).
4. Abrir `http://localhost:8080/swagger-ui.html`.
5. Renombrar el nombre de proyecto (pom y properties) y registrar los requisitos reales de EP3 según rúbrica oficial antes de implementar funcionalidades.

## Trabajo del grupo
- Hacer inventario de páginas y formularios del frontend EP2.
- Definir entidades, relaciones y validaciones de negocio.
- Diseñar endpoints y respuestas HTTP antes de conectarlos con React.
- Separar `controller`, `service`, `repository`, `model`, y `dto` cuando corresponda.
- Incorporar un módulo a la vez; evitar eliminar de golpe toda la persistencia existente en `localStorage`.

## Comunicación React-Vite
El backend escucha en `http://localhost:8080` y la aplicación Vite suele iniciar en `http://localhost:5173`.
Ejemplo de consumo (en frontend):
```javascript
const respuesta = await fetch('http://localhost:8080/api/salud');
if (!respuesta.ok) throw new Error(`HTTP ${respuesta.status}`);
const resultado = await respuesta.json();
console.log(resultado);
```

## Límites de la plantilla
- `@CrossOrigin` está restringido al origen local Vite para demostración.
- H2 es temporal y en memoria, NO se diseñó para producción.
- No incluye autenticación, autorización, dominio ni endpoints de la evaluación; deben implementarse según los requisitos reales y rúbrica EP3.
- La primera ejecución requiere internet para dependencias y Maven Wrapper.
- Verificar arranque y Swagger en un PC de laboratorio: no fue posible descargar dependencias Maven en el entorno de generación.

import io.swagger.v3.oas.models.security.SecurityScheme;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

@Configuration
public class OpenApiConfig {

    @Bean
    public OpenAPI customOpenAPI() {
        final String securitySchemeName = "bearerAuth";
        return new OpenAPI()
                .addSecurityItem(new SecurityRequirement().addList(securitySchemeName))
                .components(new Components()
                        .addSecuritySchemes(securitySchemeName,
                                new SecurityScheme()
                                        .name(securitySchemeName)
                                        .type(SecurityScheme.Type.HTTP)
                                        .scheme("bearer")
                                        .bearerFormat("JWT")));
    }
}

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.config.annotation.authentication.configuration.AuthenticationConfiguration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

import java.util.List;

@Configuration
public class SecurityConfig {

    private final JwtFilter jwtFilter;

    public SecurityConfig(JwtFilter jwtFilter) {
        this.jwtFilter = jwtFilter;
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    public AuthenticationManager authenticationManager(AuthenticationConfiguration config) throws Exception {
        return config.getAuthenticationManager();
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .csrf(csrf -> csrf.disable())
            .cors(cors -> cors.configurationSource(corsConfigurationSource()))
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/auth/login", "/swagger-ui/**", "/v3/api-docs/**", "/swagger-ui.html").permitAll()
                .requestMatchers(HttpMethod.GET, "/api/productos/**").permitAll()
                .requestMatchers("/api/productos/**").hasRole("ADMIN")
                .anyRequest().authenticated()
            )
            .addFilterBefore(jwtFilter, UsernamePasswordAuthenticationFilter.class);

        return http.build();
    }

    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration config = new CorsConfiguration();
        config.setAllowedOrigins(List.of("http://localhost:5173")); // Frontend React en Vite
        config.setAllowedMethods(List.of("GET", "POST", "PUT", "DELETE", "OPTIONS"));
        config.setAllowedHeaders(List.of("Authorization", "Content-Type"));
        
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", config);
        return source;
    }
}

package cl.duoc.ep3.security;

import cl.duoc.ep3.model.Usuario;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.Date;

@Service
public class JwtService {

    // Clave secreta fija de prueba (mínimo 256 bits para HMAC SHA-256)
    private static final String SECRET_KEY_STRING = "HuertoHogarSecretKeyJwtEP3FullStack2026DuocUCSecret";
    private static final long EXPIRATION_MINUTES = 120; // 2 horas de duración

    private final SecretKey key = Keys.hmacShaKeyFor(SECRET_KEY_STRING.getBytes(StandardCharsets.UTF_8));

    public String generarToken(Usuario usuario) {
        Instant ahora = Instant.now();
        return Jwts.builder()
                .subject(usuario.getUsername())
                .claim("rol", usuario.getRol().name())
                .issuedAt(Date.from(ahora))
                .expiration(Date.from(ahora.plusSeconds(EXPIRATION_MINUTES * 60)))
                .signWith(key)
                .compact();
    }

    public String obtenerUsername(String token) {
        return Jwts.parser()
                .verifyWith(key)
                .build()
                .parseSignedClaims(token)
                .getPayload()
                .getSubject();
    }
}

package cl.duoc.ep3.security;

import cl.duoc.ep3.repository.UsuarioRepository;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.List;

@Component
public class JwtFilter extends OncePerRequestFilter {

    private final JwtService jwtService;
    private final UsuarioRepository usuarioRepository;

    public JwtFilter(JwtService jwtService, UsuarioRepository usuarioRepository) {
        this.jwtService = jwtService;
        this.usuarioRepository = usuarioRepository;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        String header = request.getHeader("Authorization");

        if (header != null && header.startsWith("Bearer ")) {
            String token = header.substring(7).trim();
            try {
                String username = jwtService.obtenerUsername(token);
                usuarioRepository.findByUsername(username).ifPresent(usuario -> {
                    var autoridades = List.of(new SimpleGrantedAuthority("ROLE_" + usuario.getRol().name()));
                    var auth = new UsernamePasswordAuthenticationToken(username, null, autoridades);
                    SecurityContextHolder.getContext().setAuthentication(auth);
                });
            } catch (Exception ex) {
                SecurityContextHolder.clearContext();
            }
        }

        filterChain.doFilter(request, response);
    }
}

package cl.duoc.ep3.controller;

import cl.duoc.ep3.dto.AuthResponse;
import cl.duoc.ep3.dto.LoginRequest;
import cl.duoc.ep3.model.Usuario;
import cl.duoc.ep3.repository.UsuarioRepository;
import cl.duoc.ep3.security.JwtService;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.Authentication;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;

import java.util.Map;

@RestController
@RequestMapping("/auth")
public class AuthController {

    private final UsuarioRepository usuarioRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    public AuthController(UsuarioRepository usuarioRepository, PasswordEncoder passwordEncoder, JwtService jwtService) {
        this.usuarioRepository = usuarioRepository;
        this.passwordEncoder = passwordEncoder;
        this.jwtService = jwtService;
    }

    @PostMapping("/login")
    public ResponseEntity<AuthResponse> login(@Valid @RequestBody LoginRequest request) {
        Usuario usuario = usuarioRepository.findByUsername(request.username())
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Credenciales incorrectas"));

        if (!passwordEncoder.matches(request.password(), usuario.getPassword())) {
            throw new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Credenciales incorrectas");
        }

        String token = jwtService.generarToken(usuario);
        return ResponseEntity.ok(new AuthResponse(token, usuario.getUsername(), usuario.getRol().name()));
    }

    @GetMapping("/me")
    public Map<String, String> me(Authentication authentication) {
        if (authentication == null || !authentication.isAuthenticated()) {
            throw new ResponseStatusException(HttpStatus.UNAUTHORIZED, "No autenticado");
        }
        Usuario usuario = usuarioRepository.findByUsername(authentication.getName())
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Usuario no encontrado"));

        return Map.of(
            "username", usuario.getUsername(),
            "rol", usuario.getRol().name()
        );
    }
}

package cl.duoc.ep3.service;

import cl.duoc.ep3.model.Producto;
import cl.duoc.ep3.repository.ProductoRepository;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import java.util.List;

@Service
public class ProductoService {

    private final ProductoRepository repository;

    public ProductoService(ProductoRepository repository) {
        this.repository = repository;
    }

    public List<Producto> listar() {
        return repository.findAll();
    }

    public Producto buscarPorId(Long id) {
        return repository.findById(id).orElseThrow(() ->
            new ResponseStatusException(HttpStatus.NOT_FOUND, "Producto no encontrado"));
    }

    public Producto crear(Producto producto) {
        producto.setId(null);
        return repository.save(producto);
    }

    public Producto actualizar(Long id, Producto cambios) {
        Producto actual = buscarPorId(id);
        actual.setCodigo(cambios.getCodigo());
        actual.setNombre(cambios.getNombre());
        actual.setCategoria(cambios.getCategoria());
        actual.setPrecio(cambios.getPrecio());
        actual.setStock(cambios.getStock());
        actual.setUnidad(cambios.getUnidad());
        actual.setDescripcion(cambios.getDescripcion());
        return repository.save(actual);
    }

    public void eliminar(Long id) {
        Producto actual = buscarPorId(id);
        repository.delete(actual);
    }
}
